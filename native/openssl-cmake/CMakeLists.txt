cmake_minimum_required(VERSION 3.16)
project(openssl_cmake C)

# Builds OpenSSL 3.x with CMake, for OHOS. OpenSSL itself ships no CMake build
# system: its tree only has Perl templates (*.h.in) and build.info files that
# `perl Configure` turns into the generated headers and the exact compile list.
# Windows has no Perl, so instead of running Configure the CMake way is:
#
#   * the Configure output is frozen once into generated/ (export-snapshot.pl),
#     carrying the generated headers and the object -> source mapping, and
#   * this script compiles that frozen set exactly like the generated Makefile
#     did (same sources, defines, include paths and archive split).
#
# Re-generate the snapshot (native/scripts/export-openssl-snapshot.ps1) only when
# the pinned OpenSSL version changes; day-to-day builds never touch WSL/Perl.

if(NOT OHOS_ARCH)
    set(OHOS_ARCH arm64-v8a)
endif()

# Fetched upstream tree (native/scripts/fetch-sources.ps1) and the frozen
# Configure output (native/openssl-cmake/generated/).
set(OSSL_SRC "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/openssl-3.0.15"
    CACHE PATH "OpenSSL source tree")
set(OSSL_SNAPSHOT "${CMAKE_CURRENT_SOURCE_DIR}/generated"
    CACHE PATH "Frozen `perl Configure` output")

if(NOT EXISTS "${OSSL_SRC}/Configure")
    message(FATAL_ERROR "OpenSSL source not found at ${OSSL_SRC}; run native/scripts/fetch-sources.ps1 first")
endif()
if(NOT EXISTS "${OSSL_SNAPSHOT}/sources.cmake")
    message(FATAL_ERROR "OpenSSL snapshot not found under ${OSSL_SNAPSHOT}; run native/scripts/export-openssl-snapshot.ps1 first")
endif()

# Generated headers are layered on top of the source include set. The snapshot
# mirrors the tree layout (include/, providers/) so the OpenSSL include paths
# below find <openssl/*.h>, <crypto/*.h> and "prov/der_*.h" unchanged.
set(OSSL_OVERLAY "${CMAKE_CURRENT_BINARY_DIR}/overlay")
file(COPY "${OSSL_SNAPSHOT}/include/" DESTINATION "${OSSL_OVERLAY}/include")
file(COPY "${OSSL_SNAPSHOT}/providers/" DESTINATION "${OSSL_OVERLAY}/providers")

# include/openssl/configuration.h is the one generated header that is
# architecture dependent for the supported 64-bit targets (RC4_INT); pick the
# snapshot for this ABI.
if(NOT EXISTS "${OSSL_SNAPSHOT}/arch/${OHOS_ARCH}/configuration.h")
    message(FATAL_ERROR "no configuration.h for arch '${OHOS_ARCH}' in ${OSSL_SNAPSHOT}/arch")
endif()
file(COPY "${OSSL_SNAPSHOT}/arch/${OHOS_ARCH}/configuration.h"
     DESTINATION "${OSSL_OVERLAY}/include/openssl")

# A neutral buildinf.h replaces the Configure-generated one: the latter embeds
# the build machine's absolute path, compiler flags and build timestamp, none of
# which belong in a shipped library.
file(WRITE "${OSSL_OVERLAY}/crypto/buildinf.h" [=[
/*
 * Neutral build information written by native/openssl-cmake/CMakeLists.txt.
 * The Configure-generated file embeds the build machine's path/flags/date.
 */
#define PLATFORM "platform: ohos"
#define DATE "built on: reproducible"
static const char compiler_flags[] = {
    'c','o','m','p','i','l','e','r',':',' ','o','h','o','s',' ','c','l','a','n','g','\0'
};
]=])

# Source manifest, generated by export-snapshot.pl. It refers to the two roots
# above, so both must already be defined.
include("${OSSL_SNAPSHOT}/sources.cmake")

# Exactly the flags the generated Makefile used (LIB_CFLAGS/LIB_CPPFLAGS plus the
# fixed -DNDEBUG / OPENSSL_BUILDING_OPENSSL from Configure), minus anything that
# would leak a build-machine path. Optimisation is pinned (-O2) so the prebuilt
# archives do not change with the app build mode - same rule as the FreeRDP
# prebuilt libraries (doc_agent/native-libraries.md).
set(OSSL_DEFINES
    OPENSSL_BUILDING_OPENSSL
    OPENSSL_USE_NODELETE
    OPENSSL_PIC
    OPENSSL_NO_AUTOLOAD_CONFIG=1
    NDEBUG
    OPENSSLDIR="/dev/null"
    ENGINESDIR="/dev/null"
    MODULESDIR="/dev/null"
)

# Overlay paths come first so the generated headers win over the source
# templates' directory; providers includes are added once from each root
# (generated der_*.h, then the checked-in headers).
set(OSSL_INCLUDES
    "${OSSL_OVERLAY}/crypto"        # buildinf.h
    "${OSSL_OVERLAY}/include"       # generated <openssl/*.h>, <crypto/*.h>
    "${OSSL_OVERLAY}/providers/common/include"
    "${OSSL_SRC}"
    "${OSSL_SRC}/include"
    "${OSSL_SRC}/providers/common/include"
    "${OSSL_SRC}/providers/implementations/include"
)

add_library(crypto STATIC ${OSSL_CRYPTO_SRCS})
add_library(ssl STATIC ${OSSL_SSL_SRCS})

target_include_directories(crypto PUBLIC ${OSSL_INCLUDES})
target_include_directories(ssl PUBLIC ${OSSL_INCLUDES})
target_link_libraries(ssl PUBLIC crypto)

target_compile_definitions(crypto PRIVATE ${OSSL_DEFINES} STATIC_LEGACY)
target_compile_definitions(ssl PRIVATE ${OSSL_DEFINES})

# -fPIC: the archives end up inside libwinpr3.so / libhmrdp.so. -pthread matches
# the Configured OPENSSL_THREADS. --gcc-toolchain is passed by the OHOS toolchain
# to compile-only invocations, which clang flags as unused.
target_compile_options(crypto PRIVATE -fPIC -pthread -O2 -Wno-unused-command-line-argument)
target_compile_options(ssl PRIVATE -fPIC -pthread -O2 -Wno-unused-command-line-argument)

# Install the public header set (checked-in + generated) and the two archives,
# matching the layout build-native.ps1 and the app module expect unchanged.
install(DIRECTORY "${OSSL_SRC}/include/openssl/" DESTINATION include/openssl
        FILES_MATCHING PATTERN "*.h")
install(DIRECTORY "${OSSL_OVERLAY}/include/openssl/" DESTINATION include/openssl
        FILES_MATCHING PATTERN "*.h")
install(TARGETS crypto ssl ARCHIVE DESTINATION lib)

message(STATUS "HmRdp: OpenSSL (CMake) ${OHOS_ARCH}: ${CMAKE_CURRENT_SOURCE_DIR}")
message(STATUS "HmRdp:   source ${OSSL_SRC}")
message(STATUS "HmRdp:   snapshot ${OSSL_SNAPSHOT}")
